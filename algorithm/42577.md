function solution(pb) {
  const sortingPb = pb.sort();
  console.log("sortingPb: ", sortingPb);
  for (let el of sortingPb) {
    //split 때 앞에 인덱스0이 ""이면 됨
    const pb2 = sortingPb.filter((item) => item !== el);
    for (el2 of pb2) {
