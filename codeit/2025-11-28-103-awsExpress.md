```prisma
  id       Int     @id @default(autoincrement())
```

```ts
import express from "express";
import swaggerJsdoc from "swagger-jsdoc";
import swaggerUi from "swagger-ui-express";
import authRouter from "./routes/auth";

// ------------------------------------
// Express 앱 생성
// ------------------------------------

app.use(express.json());

// ------------------------------------
// Swagger 문서 옵션
// ------------------------------------

  definition: {
    openapi: "3.0.0",
    info: { title: "Simple API", version: "1.0.0" },
  },
  apis: ["./src/routes/*.ts"], // 라우터에서 JSDoc 읽기
});

// ------------------------------------
// Swagger UI 제공
// ------------------------------------
app.use("/docs", swaggerUi.serve, swaggerUi.setup(swaggerSpec));

// ------------------------------------
// 라우터 등록
// ------------------------------------
app.use("/auth", authRouter);

// ------------------------------------
// 서버 시작
// ------------------------------------
app.listen(3000, () => {
  console.log("Server running on 3000");
});
```

# ✅ 5. 인증 라우터 (회원가입 / 로그인 / 로그아웃)

`src/routes/auth.ts`

```ts
import { Router } from "express";
import { PrismaClient } from "@prisma/client";
import bcrypt from "bcrypt";
import jwt from "jsonwebtoken";

const prisma = new PrismaClient();
const router = Router();

// ------------------------------------
// 회원가입
// ------------------------------------
router.post("/signup", async (req, res) => {
  const { email, password, role } = req.body;

  // 비밀번호 해시
  const hash = await bcrypt.hash(password, 10);

  const user = await prisma.user.create({
    data: { email, password: hash, role },
  });

  res.json(user);
});

// ------------------------------------
// 로그인
// ------------------------------------
router.post("/login", async (req, res) => {
  const { email, password } = req.body;

  const user = await prisma.user.findUnique({ where: { email } });
  if (!user) return res.status(400).json({ message: "User not found" });

  const match = await bcrypt.compare(password, user.password);
  if (!match) return res.status(400).json({ message: "Wrong password" });

  // JWT 생성
  const token = jwt.sign(
    { id: user.id, role: user.role },
    "SECRET", // 실제는 .env 사용
    { expiresIn: "1h" }
  );

  res.json({ token });
});

// ------------------------------------
// 로그아웃 (간단히 토큰 제거 시뮬레이트)
// ------------------------------------
router.post("/logout", (_req, res) => {
  res.json({ message: "Logged out (client deletes JWT)" });
});

export default router;
```

# ✅ 6. 간단한 권한(ADMIN / SELLER / USER) 미들웨어

`src/middleware/role.ts`

```ts
import { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";

// ------------------------------------
// 역할 기반 인가 미들웨어
// ------------------------------------
export const requireRole = (role: string) => {
  return (req: Request, res: Response, next: NextFunction) => {
    try {
      const token = req.headers.authorization?.split(" ")[1];
      if (!token) return res.status(401).json({ message: "No token" });

      const decoded = jwt.verify(token, "SECRET") as any;

      if (decoded.role !== role)
        return res.status(403).json({ message: "Forbidden" });

      next();
    } catch {
      return res.status(401).json({ message: "Invalid token" });
    }
  };
};
```

# ✅ 7. 디스코드 진행상황 체크봇

(특정 기간 + 특정 요일 + 특정 시간 → 자동으로 "스레드" 생성)

⚠️ **Discord는 cron-like 스케줄러 사용이 가장 간단**

`npm i discord.js node-cron`

`src/bot.ts`

```ts
import { Client, GatewayIntentBits } from "discord.js";
import cron from "node-cron";

// ------------------------------------
// Discord Bot 클라이언트 생성
// ------------------------------------
const client = new Client({
  intents: [GatewayIntentBits.Guilds, GatewayIntentBits.GuildMessages],
});

// ------------------------------------
// 봇 로그인
// ------------------------------------
client.login(process.env.DISCORD_TOKEN);

// ------------------------------------
// 매주 월·수·금 오후 20:00 스레드 자동 생성
// cron 형식: "0 20 * * 1,3,5"
// ------------------------------------
client.on("ready", () => {
  console.log(`Bot logged in`);

  cron.schedule("0 20 * * 1,3,5", async () => {
    const channel = await client.channels.fetch(process.env.CHANNEL_ID!);

    if (channel?.isTextBased()) {
      await channel.threads.create({
        name: `진행상황 체크 - ${new Date().toLocaleDateString()}`,
        autoArchiveDuration: 1440, // 24시간 후 자동 보관
      });
    }
  });
});
```
